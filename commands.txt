# 2. html2text (Python)
# Post-process: convert indented code blocks to fenced with language from
# data-lang attributes. html2text doesn't support fenced blocks natively.
python -c "
import re
from bs4 import BeautifulSoup
import html2text
with open('src/docs.crpt.ru_gismt_True_API_.htm', 'r', encoding='utf-8') as f:
    html = f.read()
soup = BeautifulSoup(html, 'html.parser')
# Build a map of code text -> lang from the HTML
lang_map = {}
for pre in soup.find_all('pre', class_='highlight'):
    code = pre.find('code', attrs={'data-lang': True})
    if code:
        # Use first 60 chars of text content as key
        key = code.get_text()[:60].strip()
        lang_map[key] = code['data-lang']
# Build heading text -> id map
heading_ids = {}
for h_tag in soup.find_all(['h1','h2','h3','h4','h5','h6']):
    if h_tag.get('id'):
        heading_ids[h_tag.get_text().strip()] = h_tag['id']
# Protect complex tables from conversion
complex_tables = {}
for idx, table in enumerate(soup.find_all('table')):
    is_complex = False
    for cell in table.find_all(['td', 'th']):
        if len(cell.find_all(['p', 'ul', 'ol', 'pre'])) > 1:
            is_complex = True
            break
    if is_complex:
        placeholder = 'COMPLEX_TABLE_%d' % idx
        complex_tables[placeholder] = str(table)
        table.replace_with(placeholder)
h = html2text.HTML2Text()
h.body_width = 0
html2 = str(soup).replace('https://docs.crpt.ru/gismt/True_API/#', '#')
md = h.handle(html2)
# Replace indented code blocks with fenced ones, adding lang where possible
lines = md.split('\n')
result = []
i = 0
while i < len(lines):
    # Detect indented code block (4+ spaces or tab)
    if lines[i].startswith('    ') or lines[i].startswith('\t'):
        block = []
        while i < len(lines) and (lines[i].startswith('    ') or lines[i].startswith('\t') or lines[i].strip() == ''):
            if lines[i].strip() == '' and i + 1 < len(lines) and not (lines[i+1].startswith('    ') or lines[i+1].startswith('\t')):
                break
            block.append(lines[i])
            i += 1
        # Strip leading indent
        code_lines = []
        for l in block:
            if l.startswith('    '):
                code_lines.append(l[4:])
            elif l.startswith('\t'):
                code_lines.append(l[1:])
            else:
                code_lines.append(l)
        code_text = '\n'.join(code_lines).strip()
        # Try to find language
        key = code_text[:60].strip()
        lang = lang_map.get(key, '')
        result.append('\`\`\`' + lang)
        result.append(code_text)
        result.append('\`\`\`')
    else:
        result.append(lines[i])
        i += 1
md = '\n'.join(result)
# Insert anchor ids before headings
def insert_anchor(m):
    hashes, text = m.group(1), m.group(2).strip()
    hid = heading_ids.get(text, '')
    if hid:
        return '<a id=\"' + hid + '\"></a>\n\n' + hashes + ' ' + text
    return m.group(0)
md = re.sub(r'^(#{1,6}) (.+)$', insert_anchor, md, flags=re.MULTILINE)
# Restore complex tables as raw HTML
for placeholder, html_table in complex_tables.items():
    md = md.replace(placeholder, '\n\n' + html_table + '\n\n')
with open('mds/html2text/README.md', 'w', encoding='utf-8') as f:
    f.write(md)

# 4. Turndown (Node.js + GFM plugin)
# Pre-strip <style>/<script> tags; custom rule: strip <p> inside <td>/<th>
node -e "
const TurndownService = require('turndown');
const {gfm} = require('turndown-plugin-gfm');
const fs = require('fs');
let html = fs.readFileSync('src/docs.crpt.ru_gismt_True_API_.htm', 'utf8');
html = html.replace(/<style[^>]*>[\s\S]*?<\/style>/gi, '');
html = html.replace(/<script[^>]*>[\s\S]*?<\/script>/gi, '');
html = html.replace(/https:\/\/docs\.crpt\.ru\/gismt\/True_API\/#/g, '#');
// Build heading text -> id map
const headingIds = {};
const hRe = /<h[1-6]\s+id=\"([^\"]*)\"[^>]*>([\s\S]*?)<\/h[1-6]>/gi;
let hm;
while ((hm = hRe.exec(html)) !== null) {
  headingIds[hm[2].replace(/<[^>]*>/g, '').trim()] = hm[1];
}
const td = new TurndownService({headingStyle: 'atx', codeBlockStyle: 'fenced'});
td.use(gfm);
// Keep complex tables (cells with multiple <p>, <ul>, <ol>) as raw HTML
td.addRule('complexTable', {
  filter: function(node) {
    if (node.nodeName !== 'TABLE') return false;
    var cells = node.querySelectorAll('td, th');
    for (var i = 0; i < cells.length; i++) {
      var blocks = cells[i].querySelectorAll('p, ul, ol, pre');
      if (blocks.length > 1) return true;
    }
    return false;
  },
  replacement: function(content, node) { return '\n\n' + node.outerHTML + '\n\n'; }
});
// For simple tables: strip <p> inside cells so rows stay on one line
td.addRule('tableCellParagraph', {
  filter: function(node) {
    if (node.nodeName !== 'P' || !node.parentNode) return false;
    if (node.parentNode.nodeName !== 'TD' && node.parentNode.nodeName !== 'TH') return false;
    // Only apply to simple tables (not already handled by complexTable)
    var table = node.closest('table');
    if (!table) return false;
    var cells = table.querySelectorAll('td, th');
    for (var i = 0; i < cells.length; i++) {
      if (cells[i].querySelectorAll('p, ul, ol, pre').length > 1) return false;
    }
    return true;
  },
  replacement: function(content) { return content.trim(); }
});
let md = td.turndown(html);
// Post-process: insert anchor ids before headings
md = md.replace(/^(#{1,6}) (.+)$/gm, function(match, hashes, text) {
  const clean = text.replace(/\\\*/g, '*').replace(/\\\[/g, '[').replace(/\\\]/g, ']').trim();
  const hid = headingIds[clean] || headingIds[text.trim()];
  if (hid) return '<a id=\"' + hid + '\"></a>\n\n' + hashes + ' ' + text;
  return match;
});
fs.writeFileSync('mds/turndown/README.md', md, 'utf8');
"
