# Conversion commands for src/docs.crpt.ru_gismt_True_API_.htm -> .md
# Run from repo root

# 1. Pandoc (GFM output)
pandoc "src/docs.crpt.ru_gismt_True_API_.htm" -f html-native_divs-native_spans -t gfm --wrap=none -o "mds/pandoc/README.md"

# 2. html2text (Python)
python -c "
import html2text
h = html2text.HTML2Text()
h.body_width = 0
with open('src/docs.crpt.ru_gismt_True_API_.htm', 'r', encoding='utf-8') as f:
    html = f.read()
md = h.handle(html)
with open('mds/html2text/README.md', 'w', encoding='utf-8') as f:
    f.write(md)
"

# 3. markdownify (Python)
# Pre-strip <style> and <script> tags with BeautifulSoup before conversion
python -c "
from markdownify import markdownify as md_convert
from bs4 import BeautifulSoup
with open('src/docs.crpt.ru_gismt_True_API_.htm', 'r', encoding='utf-8') as f:
    html = f.read()
soup = BeautifulSoup(html, 'html.parser')
for tag in soup.find_all(['script', 'style']):
    tag.decompose()
md = md_convert(str(soup), heading_style='ATX')
with open('mds/markdownify/README.md', 'w', encoding='utf-8') as f:
    f.write(md)
"

# 4. Turndown (Node.js + GFM plugin)
# Pre-strip <style>/<script> tags; custom rule: strip <p> inside <td>/<th>
node -e "
const TurndownService = require('turndown');
const {gfm} = require('turndown-plugin-gfm');
const fs = require('fs');
let html = fs.readFileSync('src/docs.crpt.ru_gismt_True_API_.htm', 'utf8');
html = html.replace(/<style[^>]*>[\s\S]*?<\/style>/gi, '');
html = html.replace(/<script[^>]*>[\s\S]*?<\/script>/gi, '');
const td = new TurndownService({headingStyle: 'atx', codeBlockStyle: 'fenced'});
td.use(gfm);
td.addRule('tableCellParagraph', {
  filter: function(node) {
    return node.nodeName === 'P' && node.parentNode && (node.parentNode.nodeName === 'TD' || node.parentNode.nodeName === 'TH');
  },
  replacement: function(content) { return content.trim(); }
});
const md = td.turndown(html);
fs.writeFileSync('mds/turndown/README.md', md, 'utf8');
"
